from __future__ import annotations

TAX_RATE_PERCENT = 10

ROUNDING_MODES = ("half_up", "half_even", "floor", "ceil")


def tax_cents(
    amount_cents: int,
    rate_percent: int = TAX_RATE_PERCENT,
    rounding: str = "half_up",
) -> int:
    """Tax on an amount, rounded to a whole cent using the given rounding mode."""
    if rounding not in ROUNDING_MODES:
        raise ValueError(f"unknown rounding mode: {rounding}")
    if rate_percent < 0 or rate_percent > 100:
        raise ValueError("rate_percent must be between 0 and 100")
    numerator = amount_cents * rate_percent
    if rounding == "half_up":
        return (numerator + 50) // 100
    if rounding == "floor":
        return numerator // 100
    if rounding == "ceil":
        return -((-numerator) // 100)
    # half_even
    quotient, remainder = divmod(numerator, 100)
    if remainder > 50:
        return quotient + 1
    if remainder < 50:
        return quotient
    return quotient if quotient % 2 == 0 else quotient + 1


def format_money(cents: int) -> str:
    sign = "-" if cents < 0 else ""
    cents = abs(cents)
    return f"{sign}${cents // 100}.{cents % 100:02d}"

from __future__ import annotations

from dataclasses import dataclass

from .cart import Cart
from .pricing import ROUNDING_MODES, tax_cents


@dataclass
class Order:
    order_id: str
    lines: list[tuple[str, int, int]]  # (sku, quantity, unit price in cents)
    subtotal_cents: int
    tax_cents: int
    total_cents: int


def checkout(
    cart: Cart,
    order_id: str,
    rounding: str = "half_up",
    per_line: bool = False,
) -> Order:
    """Turn a cart into an order. The cart is emptied afterwards."""
    if rounding not in ROUNDING_MODES:
        raise ValueError(f"unknown rounding mode: {rounding}")
    if not isinstance(per_line, bool):
        raise TypeError("per_line must be a bool")
    if cart.is_empty():
        raise ValueError("cart is empty")
    lines = [(p.sku, qty, p.price_cents) for p, qty in cart.lines()]
    subtotal = cart.subtotal_cents()
    if per_line:
        tax = sum(
            tax_cents(unit_price * qty, rounding=rounding)
            for _sku, qty, unit_price in lines
        )
    else:
        tax = tax_cents(subtotal, rounding=rounding)
    order = Order(order_id, lines, subtotal, tax, subtotal + tax)
    cart.clear()
    return order
