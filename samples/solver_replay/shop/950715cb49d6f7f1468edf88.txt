from __future__ import annotations

from dataclasses import dataclass

from .cart import Cart
from .pricing import tax_cents

MAX_ORDER_ID_LENGTH = 32


@dataclass
class Order:
    order_id: str
    lines: list[tuple[str, int, int]]  # (sku, quantity, unit price in cents)
    subtotal_cents: int
    tax_cents: int
    total_cents: int


def _validate_order_id(order_id) -> None:
    if not isinstance(order_id, str):
        raise TypeError("order_id must be a string")
    if not order_id or order_id.isspace():
        raise ValueError("order_id must not be empty")
    if any(ch.isspace() for ch in order_id):
        raise ValueError("order_id must not contain whitespace")
    if len(order_id) > MAX_ORDER_ID_LENGTH:
        raise ValueError("order_id must be at most 32 characters")


def checkout(cart: Cart, order_id: str) -> Order:
    """Turn a cart into an order. The cart is emptied afterwards."""
    _validate_order_id(order_id)
    if cart.is_empty():
        raise ValueError("cart is empty")
    lines = [(p.sku, qty, p.price_cents) for p, qty in cart.lines()]
    subtotal = cart.subtotal_cents()
    tax = tax_cents(subtotal)
    order = Order(order_id, lines, subtotal, tax, subtotal + tax)
    cart.clear()
    return order
