=== TITLE ===
Validated membership windows that gate lending and cap due dates
=== INSTRUCTION ===
## Background

The `library` package lets members borrow books through `Library.lend(isbn, member_id, today)`. Memberships are currently open-ended. The library now wants memberships that are only valid inside a date window, and it wants malformed input to the lending entry points rejected with precise errors instead of failing later in confusing ways.

## Requirements

1. `Member` gains two optional fields, `valid_from` and `valid_until`, both defaulting to `None` (meaning "no bound on that side"). Existing positional construction `Member("m1", "Ana")` keeps working.
2. `Library.add_member` validates the window before doing anything else (including the duplicate-id check). A value that is not `None` must be a `datetime.date` and must not be a `datetime.datetime`; otherwise raise `TypeError` with the message `valid_from must be a date` or `valid_until must be a date` respectively (check `valid_from` first). A rejected member must not be registered.
3. If both bounds are set and `valid_from` is later than `valid_until`, `add_member` raises `ValueError("valid_from is after valid_until")`. A window where both bounds are the same day is valid.
4. `Library.lend` validates `today` first, before any lookup: it must be a `datetime.date` and not a `datetime.datetime`; otherwise raise `TypeError("today must be a date")`. This check takes precedence over the existing unknown-isbn / unknown-member `KeyError`s.
5. After the existing unknown isbn / unknown member checks, and before the "already borrowed" and "no copies available" checks, `lend` enforces the membership window. Both bounds are inclusive: if `today` is earlier than `valid_from` or later than `valid_until`, raise `ValueError` with the message `membership not active on <today in ISO format>`. A rejected lend must not create a loan.
6. A loan can never outlive the membership: when the member has a `valid_until`, the due date returned by `lend` (and used for later fines) is the earlier of `today + LOAN_DAYS` and `valid_until`. Members without `valid_until` behave exactly as before.
7. `return_book` is unchanged and is allowed after a membership has expired; the late fine is computed against the (possibly capped) due date.

## Public interface

- `library.models.Member(member_id: str, name: str, valid_from: date | None = None, valid_until: date | None = None)` (still a frozen dataclass).
- `Library.add_member(member)` raises `TypeError("valid_from must be a date")`, `TypeError("valid_until must be a date")` or `ValueError("valid_from is after valid_until")` as described above; duplicate ids still raise `ValueError("duplicate member: <id>")` afterwards.
- `Library.lend(isbn, member_id, today) -> date` raises `TypeError("today must be a date")` and `ValueError("membership not active on YYYY-MM-DD")` as described above.
- No other public names change.

## Examples

```python
lib.add_member(Member("m3", "Cy", valid_from=date(2026, 3, 1), valid_until=date(2026, 3, 5)))
lib.lend("111", "m3", date(2026, 3, 1))   # -> date(2026, 3, 5), capped by valid_until
lib.lend("222", "m3", date(2026, 3, 6))   # ValueError: membership not active on 2026-03-06
lib.lend("222", "m3", "2026-03-02")       # TypeError: today must be a date
lib.add_member(Member("m4", "Di", valid_from=date(2026, 4, 1), valid_until=date(2026, 3, 1)))
                                          # ValueError: valid_from is after valid_until
```

@dataclass(frozen=True)
class Book:

class Member:

    valid_from: date | None = None

=== FILE solution library/lending.py ===
from __future__ import annotations

from datetime import date, datetime, timedelta

from .models import Book, Member

LOAN_DAYS = 14
FINE_CENTS_PER_DAY = 50


def _is_plain_date(value: object) -> bool:
    return isinstance(value, date) and not isinstance(value, datetime)


class Library:
    """Books, members and who has borrowed what. Dates are always passed in."""

    def __init__(self) -> None:
        self._books: dict[str, Book] = {}
        self._members: dict[str, Member] = {}
        self._loans: dict[tuple[str, str], date] = {}  # (isbn, member_id) -> due date

    def add_book(self, book: Book) -> None:
        if book.isbn in self._books:
            raise ValueError(f"duplicate isbn: {book.isbn}")
        self._books[book.isbn] = book

    def add_member(self, member: Member) -> None:
        if member.valid_from is not None and not _is_plain_date(member.valid_from):
            raise TypeError("valid_from must be a date")
        if member.valid_until is not None and not _is_plain_date(member.valid_until):
            raise TypeError("valid_until must be a date")
        if (
            member.valid_from is not None
            and member.valid_until is not None
            and member.valid_from > member.valid_until
        ):
            raise ValueError("valid_from is after valid_until")
        if member.member_id in self._members:
            raise ValueError(f"duplicate member: {member.member_id}")
        self._members[member.member_id] = member

    def copies_available(self, isbn: str) -> int:
        book = self._book(isbn)
        on_loan = sum(1 for (i, _m) in self._loans if i == isbn)
        return book.copies - on_loan

    def lend(self, isbn: str, member_id: str, today: date) -> date:
        """Lend a book and return its due date."""
        if not _is_plain_date(today):
            raise TypeError("today must be a date")
        self._book(isbn)
        member = self._member(member_id)
        if (member.valid_from is not None and today < member.valid_from) or (
            member.valid_until is not None and today > member.valid_until
        ):
            raise ValueError(f"membership not active on {today.isoformat()}")
        if (isbn, member_id) in self._loans:
            raise ValueError("already borrowed")
        if self.copies_available(isbn) <= 0:
            raise ValueError("no copies available")
        due = today + timedelta(days=LOAN_DAYS)
        if member.valid_until is not None:
            due = min(due, member.valid_until)
        self._loans[(isbn, member_id)] = due
        return due

    def return_book(self, isbn: str, member_id: str, today: date) -> int:
        """Take a book back and return the late fine in cents."""
        key = (isbn, member_id)
        if key not in self._loans:
            raise KeyError("no such loan")
        due = self._loans.pop(key)
        late_days = max(0, (today - due).days)
        return late_days * FINE_CENTS_PER_DAY

    def loans_of(self, member_id: str) -> list[str]:
        self._member(member_id)
        return sorted(i for (i, m) in self._loans if m == member_id)

    def _book(self, isbn: str) -> Book:
        try:
            return self._books[isbn]
        except KeyError:
            raise KeyError(f"unknown isbn: {isbn}") from None

    def _member(self, member_id: str) -> Member:
        try:
            return self._members[member_id]
        except KeyError:
            raise KeyError(f"unknown member: {member_id}") from None

from datetime import date, datetime, timedelta

import pytest

from library import FINE_CENTS_PER_DAY, LOAN_DAYS, Member


def test_member_window_fields_default_to_none():
    m = Member("x", "Xi")
    assert m.valid_from is None
    assert m.valid_until is None


def test_add_member_rejects_non_date_bounds(lib):
    with pytest.raises(TypeError, match="valid_from must be a date"):
        lib.add_member(Member("m3", "Cy", valid_from="2026-03-01"))
    with pytest.raises(TypeError, match="valid_until must be a date"):
        lib.add_member(Member("m3", "Cy", valid_until=20260301))
    with pytest.raises(TypeError, match="valid_from must be a date"):
        lib.add_member(Member("m3", "Cy", valid_from=datetime(2026, 3, 1, 9, 0)))
    with pytest.raises(TypeError, match="valid_until must be a date"):
        lib.add_member(Member("m3", "Cy", valid_until=datetime(2026, 3, 1, 9, 0)))


def test_valid_from_is_checked_before_valid_until(lib):
    with pytest.raises(TypeError, match="valid_from must be a date"):
        lib.add_member(Member("m3", "Cy", valid_from="x", valid_until="y"))


def test_add_member_rejects_reversed_window_and_does_not_register(lib, today):
    with pytest.raises(ValueError, match="valid_from is after valid_until"):
        lib.add_member(
            Member("m3", "Cy", valid_from=today + timedelta(days=1), valid_until=today)
        )
    with pytest.raises(KeyError):
        lib.loans_of("m3")
    # the id is still free, so a corrected member can be registered
    lib.add_member(Member("m3", "Cy", valid_from=today, valid_until=today))
    assert lib.loans_of("m3") == []


def test_validation_runs_before_duplicate_check(lib, today):
    with pytest.raises(ValueError, match="valid_from is after valid_until"):
        lib.add_member(
            Member("m1", "Dup", valid_from=today + timedelta(days=2), valid_until=today)
        )


def test_single_day_window_is_valid(lib, today):
    lib.add_member(Member("m3", "Cy", valid_from=today, valid_until=today))
    assert lib.lend("111", "m3", today) == today


def test_lend_rejects_non_date_today(lib, today):
    with pytest.raises(TypeError, match="today must be a date"):
        lib.lend("111", "m1", "2026-03-01")
    with pytest.raises(TypeError, match="today must be a date"):
        lib.lend("111", "m1", datetime(2026, 3, 1, 12, 0))
    assert lib.loans_of("m1") == []


def test_today_type_check_precedes_lookups(lib):
    with pytest.raises(TypeError, match="today must be a date"):
        lib.lend("999", "nobody", None)


def test_lend_before_window_start_is_rejected(lib, today):
    lib.add_member(
        Member("m3", "Cy", valid_from=today + timedelta(days=1), valid_until=today + timedelta(days=30))
    )
    with pytest.raises(ValueError, match="membership not active on 2026-03-01"):
        lib.lend("111", "m3", today)
    assert lib.loans_of("m3") == []
    assert lib.copies_available("111") == 2


def test_lend_window_bounds_are_inclusive(lib, today):
    start = today
    end = today + timedelta(days=5)
    lib.add_member(Member("m3", "Cy", valid_from=start, valid_until=end))
    assert lib.lend("111", "m3", start) == end  # first day is allowed
    lib.return_book("111", "m3", start)
    assert lib.lend("111", "m3", end) == end  # last day is allowed
    lib.return_book("111", "m3", end)
    with pytest.raises(ValueError, match="membership not active on 2026-03-07"):
        lib.lend("111", "m3", end + timedelta(days=1))


def test_open_ended_bounds(lib, today):
    lib.add_member(Member("m3", "Cy", valid_from=today))
    lib.add_member(Member("m4", "Di", valid_until=today + timedelta(days=100)))
    with pytest.raises(ValueError, match="membership not active"):
        lib.lend("111", "m3", today - timedelta(days=1))
    assert lib.lend("111", "m3", today + timedelta(days=500)) == today + timedelta(
        days=500 + LOAN_DAYS
    )
    assert lib.lend("222", "m4", today - timedelta(days=400)) == today - timedelta(
        days=400 - LOAN_DAYS
    )


def test_due_date_is_capped_by_valid_until(lib, today):
    lib.add_member(Member("m3", "Cy", valid_until=today + timedelta(days=5)))
    assert lib.lend("111", "m3", today) == today + timedelta(days=5)


def test_due_date_not_capped_when_membership_outlasts_loan(lib, today):
    lib.add_member(Member("m3", "Cy", valid_until=today + timedelta(days=LOAN_DAYS)))
    lib.add_member(Member("m4", "Di", valid_until=today + timedelta(days=365)))
    assert lib.lend("111", "m3", today) == today + timedelta(days=LOAN_DAYS)
    assert lib.lend("111", "m4", today) == today + timedelta(days=LOAN_DAYS)


def test_fine_uses_capped_due_date_and_return_works_after_expiry(lib, today):
    end = today + timedelta(days=5)
    lib.add_member(Member("m3", "Cy", valid_until=end))
    lib.lend("111", "m3", today)
    # the membership expired on `end`; returning is still allowed and is fined from the capped due date
    assert lib.return_book("111", "m3", end + timedelta(days=3)) == 3 * FINE_CENTS_PER_DAY
    assert lib.loans_of("m3") == []


def test_window_check_precedes_already_borrowed(lib, today):
    end = today + timedelta(days=2)
    lib.add_member(Member("m3", "Cy", valid_until=end))
    lib.lend("111", "m3", today)
    with pytest.raises(ValueError, match="membership not active on 2026-03-04"):
        lib.lend("111", "m3", end + timedelta(days=1))
    with pytest.raises(ValueError, match="already borrowed"):
        lib.lend("111", "m3", today)
