=== TITLE ===
Stack high-demand surcharge and reduced-rate discount on late fines with a daily cap
=== INSTRUCTION ===
## Background

Late fines are currently a flat `FINE_CENTS_PER_DAY` (50 cents) per late day. The library wants two independent adjustments to the daily fine rate that can both apply to the same return:

* a **high-demand** flag on a book (popular titles are fined more heavily), and
* a **reduced-rate** flag on a member (e.g. students and seniors are fined less).

When both apply they must be combined in a fixed order, and the resulting daily rate may never exceed a maximum.

## Requirements

1. Add three module-level constants in `library/lending.py`, also exported from the `library` package: `DEMAND_MULTIPLIER = 3`, `REDUCED_MEMBER_DISCOUNT_CENTS = 20` and `MAX_DAILY_RATE_CENTS = 140`.
2. `Library.set_high_demand(isbn, flag)` marks (`flag` true) or unmarks (`flag` false) a book as high-demand. Books are not high-demand by default. An unknown ISBN raises `KeyError`.
3. `Library.set_reduced_rate(member_id, flag)` marks (`flag` true) or unmarks (`flag` false) a member as reduced-rate. Members are not reduced-rate by default. An unknown member raises `KeyError`.
4. `Library.daily_fine_rate(isbn, member_id)` returns the daily fine in cents for that book and member. It starts from `FINE_CENTS_PER_DAY` and applies the rules in this exact order:
   1. if the book is high-demand, multiply the rate by `DEMAND_MULTIPLIER`;
   2. then, if the member is reduced-rate, subtract `REDUCED_MEMBER_DISCOUNT_CENTS`;
   3. finally, cap the result at `MAX_DAILY_RATE_CENTS` (the cap is applied last, after the discount).
   An unknown ISBN or unknown member raises `KeyError`.
5. `Library.return_book` must compute the fine as `late_days * daily_fine_rate(isbn, member_id)`, using the flags as they are at the moment of return. Late days are computed exactly as before (never negative), so an on-time return costs 0 whatever the flags. The existing errors (`KeyError("no such loan")`) are unchanged, and fines for books and members with no flags set stay exactly as before.
6. Flags are independent per book and per member: a flag set on one book or member never changes the rate for another. Flags persist after a book is returned.

## Public interface

* `library.lending.DEMAND_MULTIPLIER: int` (3), `REDUCED_MEMBER_DISCOUNT_CENTS: int` (20), `MAX_DAILY_RATE_CENTS: int` (140); all three importable from `library` as well.
* `Library.set_high_demand(self, isbn: str, flag: bool) -> None`
* `Library.set_reduced_rate(self, member_id: str, flag: bool) -> None`
* `Library.daily_fine_rate(self, isbn: str, member_id: str) -> int`
* Unknown ISBN or member id raises `KeyError` (same lookups as the existing code).
* `Library.return_book(isbn, member_id, today) -> int` keeps its signature and returns the fine in cents.

## Examples

```python
lib.set_high_demand("111", True)
lib.daily_fine_rate("111", "m2")   # 150 capped to 140
lib.set_reduced_rate("m1", True)
lib.daily_fine_rate("222", "m1")   # 50 - 20 = 30
lib.daily_fine_rate("111", "m1")   # 150 - 20 = 130 (under the cap)
```

If "111" is high-demand and "m1" is reduced-rate, returning it 3 days late costs `3 * 130 = 390` cents. If only the book is high-demand, it costs `3 * 140 = 420` cents. If neither flag is set, it costs `3 * 50 = 150` cents.
=== FILE solution library/lending.py ===
from __future__ import annotations

from datetime import date, timedelta

from .models import Book, Member

LOAN_DAYS = 14
FINE_CENTS_PER_DAY = 50
DEMAND_MULTIPLIER = 3
REDUCED_MEMBER_DISCOUNT_CENTS = 20
MAX_DAILY_RATE_CENTS = 140


class Library:
    """Books, members and who has borrowed what. Dates are always passed in."""

    def __init__(self) -> None:
        self._books: dict[str, Book] = {}
        self._members: dict[str, Member] = {}
        self._loans: dict[tuple[str, str], date] = {}  # (isbn, member_id) -> due date
        self._high_demand: set[str] = set()
        self._reduced_rate: set[str] = set()

    def add_book(self, book: Book) -> None:
        if book.isbn in self._books:
            raise ValueError(f"duplicate isbn: {book.isbn}")
        self._books[book.isbn] = book

    def add_member(self, member: Member) -> None:
        if member.member_id in self._members:
            raise ValueError(f"duplicate member: {member.member_id}")
        self._members[member.member_id] = member

    def copies_available(self, isbn: str) -> int:
        book = self._book(isbn)
        on_loan = sum(1 for (i, _m) in self._loans if i == isbn)
        return book.copies - on_loan

    def lend(self, isbn: str, member_id: str, today: date) -> date:
        """Lend a book and return its due date."""
        self._book(isbn)
        self._member(member_id)
        if (isbn, member_id) in self._loans:
            raise ValueError("already borrowed")
        if self.copies_available(isbn) <= 0:
            raise ValueError("no copies available")
        due = today + timedelta(days=LOAN_DAYS)
        self._loans[(isbn, member_id)] = due
        return due

    def set_high_demand(self, isbn: str, flag: bool) -> None:
        """Mark or unmark a book as high-demand (heavier late fines)."""
        self._book(isbn)
        if flag:
            self._high_demand.add(isbn)
        else:
            self._high_demand.discard(isbn)

    def set_reduced_rate(self, member_id: str, flag: bool) -> None:
        """Mark or unmark a member as reduced-rate (lighter late fines)."""
        self._member(member_id)
        if flag:
            self._reduced_rate.add(member_id)
        else:
            self._reduced_rate.discard(member_id)

    def daily_fine_rate(self, isbn: str, member_id: str) -> int:
        """Daily late fine in cents: surcharge first, then discount, then the cap."""
        self._book(isbn)
        self._member(member_id)
        rate = FINE_CENTS_PER_DAY
        if isbn in self._high_demand:
            rate *= DEMAND_MULTIPLIER
        if member_id in self._reduced_rate:
            rate -= REDUCED_MEMBER_DISCOUNT_CENTS
        return min(rate, MAX_DAILY_RATE_CENTS)

    def return_book(self, isbn: str, member_id: str, today: date) -> int:
        """Take a book back and return the late fine in cents."""
        key = (isbn, member_id)
        if key not in self._loans:
            raise KeyError("no such loan")
        due = self._loans.pop(key)
        late_days = max(0, (today - due).days)
        return late_days * self.daily_fine_rate(isbn, member_id)

    def loans_of(self, member_id: str) -> list[str]:
        self._member(member_id)
        return sorted(i for (i, m) in self._loans if m == member_id)

    def _book(self, isbn: str) -> Book:
        try:
            return self._books[isbn]
        except KeyError:
            raise KeyError(f"unknown isbn: {isbn}") from None

    def _member(self, member_id: str) -> Member:
        try:
            return self._members[member_id]
        except KeyError:
            raise KeyError(f"unknown member: {member_id}") from None

=== FILE test tests/test_tf_rule_stacking_05.py ===
from datetime import timedelta

import pytest

import library
from library import Book, Library, Member


def _late_return(lib, isbn, member_id, today, late_days):
    due = lib.lend(isbn, member_id, today)
    return lib.return_book(isbn, member_id, due + timedelta(days=late_days))


def test_constants_exported():
    assert library.DEMAND_MULTIPLIER == 3
    assert library.REDUCED_MEMBER_DISCOUNT_CENTS == 20
    assert library.MAX_DAILY_RATE_CENTS == 140


def test_no_flags_keeps_base_rate(lib, today):
    assert lib.daily_fine_rate("111", "m1") == 50
    assert _late_return(lib, "111", "m1", today, 3) == 150


def test_only_reduced_member(lib, today):
    lib.set_reduced_rate("m1", True)
    assert lib.daily_fine_rate("111", "m1") == 30
    assert _late_return(lib, "111", "m1", today, 3) == 90


def test_only_high_demand_is_capped(lib, today):
    lib.set_high_demand("111", True)
    assert lib.daily_fine_rate("111", "m1") == 140
    assert _late_return(lib, "111", "m1", today, 3) == 420


def test_both_apply_in_order_then_cap(lib, today):
    lib.set_high_demand("111", True)
    lib.set_reduced_rate("m1", True)
    # (50 * 3) - 20 = 130, under the cap; the other order would give 90
    assert lib.daily_fine_rate("111", "m1") == 130
    assert _late_return(lib, "111", "m1", today, 3) == 390


def test_on_time_return_is_free_with_flags(lib, today):
    lib.set_high_demand("111", True)
    lib.set_reduced_rate("m1", True)
    assert _late_return(lib, "111", "m1", today, 0) == 0
    assert _late_return(lib, "111", "m2", today, -5) == 0


def test_flags_are_independent(lib, today):
    lib.set_high_demand("111", True)
    lib.set_reduced_rate("m1", True)
    assert lib.daily_fine_rate("222", "m1") == 30
    assert lib.daily_fine_rate("222", "m2") == 50
    assert lib.daily_fine_rate("111", "m2") == 140
    assert _late_return(lib, "111", "m2", today, 2) == 280


def test_unsetting_flags_restores_rate(lib):
    lib.set_high_demand("111", True)
    lib.set_reduced_rate("m1", True)
    lib.set_high_demand("111", False)
    assert lib.daily_fine_rate("111", "m1") == 30
    lib.set_reduced_rate("m1", False)
    assert lib.daily_fine_rate("111", "m1") == 50


def test_flags_read_at_return_time_and_persist(lib, today):
    due = lib.lend("111", "m1", today)
    lib.set_high_demand("111", True)
    assert lib.return_book("111", "m1", due + timedelta(days=1)) == 140
    assert lib.daily_fine_rate("111", "m1") == 140


def test_unknown_ids_raise_key_error(lib):
    with pytest.raises(KeyError):
        lib.set_high_demand("999", True)
    with pytest.raises(KeyError):
        lib.set_reduced_rate("nobody", True)
    with pytest.raises(KeyError):
        lib.daily_fine_rate("999", "m1")
    with pytest.raises(KeyError):
        lib.daily_fine_rate("111", "nobody")


def test_return_without_loan_still_key_error(lib, today):
    lib.set_high_demand("111", True)
    with pytest.raises(KeyError):
        lib.return_book("111", "m1", today)
