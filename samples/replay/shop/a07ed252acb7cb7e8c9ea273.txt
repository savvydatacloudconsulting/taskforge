=== TITLE ===
Selectable tax rounding modes with per-line tax and strict validation at checkout
=== INSTRUCTION ===
## Background
Tax is currently always rounded half up and always computed once on the whole cart subtotal. Merchants need to pick the rounding rule, and some need tax computed per cart line (each line rounded on its own, then summed). Because a bad rounding setting must never silently change money, invalid settings have to be rejected before anything is charged or the cart is emptied.

## Requirements
1. `tax_cents` accepts a new keyword argument `rounding` (default `"half_up"`) selecting how the exact tax `amount_cents * rate_percent / 100` is rounded to a whole cent. The supported modes are:
   - `"half_up"`: nearest cent; an exact half rounds up (this is the existing behaviour).
   - `"half_even"`: nearest cent; an exact half rounds to the even cent (banker's rounding).
   - `"floor"`: always round down.
   - `"ceil"`: always round up (an exact whole-cent result is unchanged).
2. Calling `tax_cents` without `rounding` must return exactly what it returned before.
3. `tax_cents` raises `ValueError("unknown rounding mode: <value>")` if `rounding` is not one of the four modes, where `<value>` is the given value formatted with `str`.
4. `tax_cents` raises `ValueError("rate_percent must be between 0 and 100")` if `rate_percent` is below 0 or above 100 (0 and 100 themselves are valid). If both `rounding` and `rate_percent` are invalid, the rounding error is the one raised.
5. `checkout` accepts new keyword arguments `rounding` (default `"half_up"`) and `per_line` (default `False`).
6. With `per_line=False` the order tax is `tax_cents(subtotal, rounding=rounding)`. With `per_line=True` the order tax is the sum, over the cart lines, of `tax_cents(unit_price * quantity, rounding=rounding)`. `subtotal_cents` and `total_cents` (subtotal plus tax) follow as before.
7. `checkout` raises `ValueError("unknown rounding mode: <value>")` for an unsupported `rounding`, and `TypeError("per_line must be a bool")` if `per_line` is not a `bool`. These checks happen before anything else, so they are raised even when the cart is empty (the rounding check is made first), and in every rejected case the cart is left untouched.
8. A valid `checkout` call still empties the cart afterwards, and an empty cart with valid settings still raises `ValueError("cart is empty")`.

## Public interface
- `shop.pricing.ROUNDING_MODES`: tuple `("half_up", "half_even", "floor", "ceil")`.
- `shop.pricing.tax_cents(amount_cents, rate_percent=TAX_RATE_PERCENT, rounding="half_up") -> int`
- `shop.checkout.checkout(cart, order_id, rounding="half_up", per_line=False) -> Order`
- Errors: `ValueError("unknown rounding mode: X")`, `ValueError("rate_percent must be between 0 and 100")`, `TypeError("per_line must be a bool")`.

## Examples
At the default 10% rate, tax on 5 cents is exactly 0.5 cents: `half_up` gives 1, `half_even` gives 0, `floor` gives 0, `ceil` gives 1. Tax on 15 cents is exactly 1.5: `half_up` 2, `half_even` 2, `floor` 1, `ceil` 2. Tax on 25 cents (2.5): `half_up` 3, `half_even` 2. Tax on 1 cent (0.1): `half_up` 0, `ceil` 1. Tax on 0 cents is 0 in every mode.

`tax_cents(3, 50, rounding="half_even")` is 2 (1.5 rounds to the even 2).

A cart holding one 5-cent item and one 1-cent item, checked out with `rounding="ceil"`: with `per_line=False` the subtotal is 6, tax is `ceil(0.6)` = 1; with `per_line=True` tax is `ceil(0.5) + ceil(0.1)` = 2.

`checkout(cart, "o1", rounding="banker")` raises `ValueError("unknown rounding mode: banker")` and the cart still holds its items.
=== FILE solution shop/pricing.py ===
from __future__ import annotations

TAX_RATE_PERCENT = 10
ROUNDING_MODES = ("half_up", "half_even", "floor", "ceil")


def check_rounding(rounding: str) -> None:
    if rounding not in ROUNDING_MODES:
        raise ValueError(f"unknown rounding mode: {rounding}")


def tax_cents(
    amount_cents: int, rate_percent: int = TAX_RATE_PERCENT, rounding: str = "half_up"
) -> int:
    """Tax on an amount, rounded to a whole cent using the chosen rounding mode."""
    check_rounding(rounding)
    if not 0 <= rate_percent <= 100:
        raise ValueError("rate_percent must be between 0 and 100")
    quotient, remainder = divmod(amount_cents * rate_percent, 100)
    if rounding == "floor":
        return quotient
    if rounding == "ceil":
        return quotient + (1 if remainder else 0)
    if remainder * 2 != 100:
        return quotient + (1 if remainder * 2 > 100 else 0)
    if rounding == "half_up":
        return quotient + 1
    return quotient + quotient % 2


def format_money(cents: int) -> str:
    sign = "-" if cents < 0 else ""
    cents = abs(cents)
    return f"{sign}${cents // 100}.{cents % 100:02d}"
=== FILE solution shop/checkout.py ===
from __future__ import annotations

from dataclasses import dataclass

from .cart import Cart
from .pricing import check_rounding, tax_cents


@dataclass
class Order:
    order_id: str
    lines: list[tuple[str, int, int]]  # (sku, quantity, unit price in cents)
    subtotal_cents: int
    tax_cents: int
    total_cents: int


def checkout(
    cart: Cart, order_id: str, rounding: str = "half_up", per_line: bool = False
) -> Order:
    """Turn a cart into an order. The cart is emptied afterwards."""
    check_rounding(rounding)
    if not isinstance(per_line, bool):
        raise TypeError("per_line must be a bool")
    if cart.is_empty():
        raise ValueError("cart is empty")
    lines = [(p.sku, qty, p.price_cents) for p, qty in cart.lines()]
    subtotal = cart.subtotal_cents()
    if per_line:
        tax = sum(tax_cents(price * qty, rounding=rounding) for _, qty, price in lines)
    else:
        tax = tax_cents(subtotal, rounding=rounding)
    order = Order(order_id, lines, subtotal, tax, subtotal + tax)
    cart.clear()
    return order

import pytest

from shop import Cart, Catalog, Product, checkout, tax_cents
from shop.pricing import ROUNDING_MODES


@pytest.fixture
def small_cart():
    catalog = Catalog([
        Product("A", "Five", 5),
        Product("B", "Five too", 5),
        Product("C", "One", 1),
        Product("D", "Fifteen", 15),
    ])
    return Cart(catalog)


def test_modes_constant():
    assert ROUNDING_MODES == ("half_up", "half_even", "floor", "ceil")


@pytest.mark.parametrize(
    "amount, mode, expected",
    [
        (5, "half_up", 1), (5, "half_even", 0), (5, "floor", 0), (5, "ceil", 1),
        (15, "half_up", 2), (15, "half_even", 2), (15, "floor", 1), (15, "ceil", 2),
        (25, "half_up", 3), (25, "half_even", 2),
        (1, "half_up", 0), (1, "half_even", 0), (1, "floor", 0), (1, "ceil", 1),
        (0, "half_up", 0), (0, "half_even", 0), (0, "floor", 0), (0, "ceil", 0),
        (10, "ceil", 1), (10, "floor", 1),
    ],
)
def test_tax_modes_at_default_rate(amount, mode, expected):
    assert tax_cents(amount, rounding=mode) == expected


def test_custom_rate_with_rounding():
    assert tax_cents(3, 50, rounding="half_even") == 2
    assert tax_cents(3, 50, rounding="half_up") == 2
    assert tax_cents(1, 50, rounding="half_even") == 0
    assert tax_cents(200, 100, rounding="ceil") == 200
    assert tax_cents(200, 0, rounding="ceil") == 0


def test_default_rounding_unchanged():
    assert tax_cents(5) == 1
    assert tax_cents(14) == 1
    assert tax_cents(15) == 2


def test_tax_rejects_unknown_rounding():
    with pytest.raises(ValueError) as exc:
        tax_cents(100, rounding="banker")
    assert str(exc.value) == "unknown rounding mode: banker"


@pytest.mark.parametrize("rate", [-1, 101])
def test_tax_rejects_bad_rate(rate):
    with pytest.raises(ValueError) as exc:
        tax_cents(100, rate)
    assert str(exc.value) == "rate_percent must be between 0 and 100"


def test_rounding_error_wins_over_rate_error():
    with pytest.raises(ValueError) as exc:
        tax_cents(100, 500, rounding="nope")
    assert str(exc.value) == "unknown rounding mode: nope"


def test_checkout_whole_vs_per_line(small_cart):
    small_cart.add("A")
    small_cart.add("B")
    order = checkout(small_cart, "o1")
    assert (order.subtotal_cents, order.tax_cents, order.total_cents) == (10, 1, 11)

    small_cart.add("A")
    small_cart.add("B")
    order = checkout(small_cart, "o2", per_line=True)
    assert (order.subtotal_cents, order.tax_cents, order.total_cents) == (10, 2, 12)


def test_checkout_per_line_with_modes(small_cart):
    small_cart.add("A")
    small_cart.add("C")
    order = checkout(small_cart, "o3", rounding="ceil", per_line=True)
    assert order.tax_cents == 2
    assert order.total_cents == 8

    small_cart.add("A")
    small_cart.add("C")
    order = checkout(small_cart, "o4", rounding="ceil")
    assert order.tax_cents == 1

    small_cart.add("A")
    small_cart.add("B")
    order = checkout(small_cart, "o5", rounding="half_even", per_line=True)
    assert order.tax_cents == 0


def test_checkout_per_line_uses_line_quantity(small_cart):
    small_cart.add("D", 3)  # line amount 45 -> tax 4.5
    order = checkout(small_cart, "o6", rounding="half_even", per_line=True)
    assert order.tax_cents == 4
    small_cart.add("D", 3)
    order = checkout(small_cart, "o7", rounding="half_up", per_line=True)
    assert order.tax_cents == 5


def test_checkout_bad_rounding_leaves_cart(small_cart):
    small_cart.add("D", 2)
    with pytest.raises(ValueError) as exc:
        checkout(small_cart, "o8", rounding="banker")
    assert str(exc.value) == "unknown rounding mode: banker"
    assert small_cart.quantity("D") == 2


@pytest.mark.parametrize("value", [1, 0, "yes", None])
def test_checkout_bad_per_line_leaves_cart(small_cart, value):
    small_cart.add("A")
    with pytest.raises(TypeError) as exc:
        checkout(small_cart, "o9", per_line=value)
    assert str(exc.value) == "per_line must be a bool"
    assert small_cart.quantity("A") == 1


def test_checkout_validation_before_empty_check(small_cart):

        checkout(small_cart, "o10", rounding="banker")
    assert str(exc.value) == "unknown rounding mode: banker"
    with pytest.raises(TypeError):
        checkout(small_cart, "o11", per_line="no")
    with pytest.raises(ValueError) as exc:
        checkout(small_cart, "o12")
    assert str(exc.value) == "cart is empty"


def test_checkout_still_empties_cart(small_cart):
    small_cart.add("D")
    checkout(small_cart, "o13", rounding="floor", per_line=True)
    assert small_cart.is_empty()
